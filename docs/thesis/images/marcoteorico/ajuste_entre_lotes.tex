% Dos niveles de control: lazo dentro del lote (dos velocidades y corte) y
% ajuste de parámetros entre lotes con el error final del lote anterior.
\begin{tikzpicture}[
    >=Latex,
    font=\small,
    bloque/.style={rectangle, draw, rounded corners=2pt, minimum height=1.0cm,
        text width=2.6cm, align=center, inner sep=3pt},
    flecha/.style={-{Latex[width=6pt,length=5pt]}, thick},
    senal/.style={-{Latex[width=6pt,length=5pt]}, thick, dashed},
    nota/.style={font=\footnotesize, align=center}
]

% Lazo dentro del lote
\node[bloque] (ctrl) {Controlador:\\dos velocidades\\y corte};
\node[bloque, right=1.0cm of ctrl] (alim) {Alimentador};
\node[bloque, right=1.0cm of alim] (celda) {Recipiente sobre\\celda de carga};
\draw[flecha] (ctrl) -- (alim);
\draw[flecha] (alim) -- (celda);
\draw[senal] (celda.north) -- ++(0,0.5) -| (ctrl.north)
    node[pos=0.25, above, nota] {Masa durante el lote};

% Consigna
\node[nota, left=0.9cm of ctrl] (cons) {Consigna};
\draw[flecha] (cons) -- (ctrl);

% Ajuste entre lotes
\node[bloque, below=1.4cm of alim, text width=4.4cm] (ajuste)
    {Ajuste del adelanto y del\\cambio de velocidad};
\draw[senal] (celda.south) |- (ajuste.east);
\draw[senal] (ajuste.west) -| (ctrl.south);
\node[nota, anchor=west] at ([xshift=2pt, yshift=-0.7cm]celda.south)
    {Error final\\del lote $k$};
\node[nota, anchor=east] at ([xshift=-2pt, yshift=-0.7cm]ctrl.south)
    {Parámetros\\del lote $k+1$};

\end{tikzpicture}
